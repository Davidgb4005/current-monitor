#define AP_SSID "ESP32_AP"
#define AP_PASS "12345678"
#define MAX_STA_CONN 4

static const char *TAG = "HTTP_SERVER";


void init_spiffs() {
    esp_vfs_spiffs_conf_t conf = {
        .base_path = "/spiffs",   // mount point
        .partition_label = "spiffs",
        .max_files = 5,           // max files open at once
        .format_if_mount_failed = true
    };

    esp_err_t ret = esp_vfs_spiffs_register(&conf);
    if (ret != ESP_OK) {
        if (ret == ESP_FAIL) {
            ESP_LOGE("SPIFFS", "Failed to mount or format filesystem");
        } else if (ret == ESP_ERR_NOT_FOUND) {
            ESP_LOGE("SPIFFS", "SPIFFS partition not found");
        } else {
            ESP_LOGE("SPIFFS", "Failed to initialize SPIFFS (%s)", esp_err_to_name(ret));
        }
        return;
    }

    size_t total = 0, used = 0;
    esp_spiffs_info("spiffs", &total, &used);
    ESP_LOGI("SPIFFS", "Partition size: total: %d, used: %d", total, used);
}

void wifi_init(void)
{
    // Initialize NVS
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND)
    {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);

    // Initialize TCP/IP
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    // Create default Wi-Fi AP
    esp_netif_create_default_wifi_ap();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    wifi_config_t wifi_config = {};
    wifi_config.ap.ssid_len = strlen(AP_SSID);
    wifi_config.ap.max_connection = MAX_STA_CONN;
    wifi_config.ap.authmode = WIFI_AUTH_WPA_WPA2_PSK;

    // Copy SSID and password into fixed-size arrays
    strncpy((char *)wifi_config.ap.ssid, AP_SSID, sizeof(wifi_config.ap.ssid));
    strncpy((char *)wifi_config.ap.password, AP_PASS, sizeof(wifi_config.ap.password));

    // If password is empty, switch to open auth
    if (strlen(AP_PASS) == 0)
    {
        wifi_config.ap.authmode = WIFI_AUTH_OPEN;
    }

    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_AP)); // Set ESP32 as AP
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "AP started. SSID:%s password:%s", AP_SSID, AP_PASS);
}

esp_err_t css_get_handler(httpd_req_t *req)
{
    FILE* f = fopen("/spiffs/style.css", "r");
    if (!f) {
        httpd_resp_send_404(req);
        return ESP_FAIL;
    }

    char buf[256];
    size_t read_bytes;
    httpd_resp_set_type(req, "text/css");

    while ((read_bytes = fread(buf, 1, sizeof(buf), f)) > 0) {
        httpd_resp_send_chunk(req, buf, read_bytes);
    }
    fclose(f);
    httpd_resp_send_chunk(req, NULL, 0); // signal end
    return ESP_OK;
}
esp_err_t alternator_get_handler(httpd_req_t *req)
{
    const char *resp_str = "{\"currentMax\": 111, \"voltageMax\": 13.6,\"current\": 100, \"voltage\": 12.5}";
    httpd_resp_set_type(req, "application/json");
    httpd_resp_send(req, resp_str, HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}
esp_err_t alternator_html_get_handler(httpd_req_t *req)
{
    FILE* f = fopen("/spiffs/alternator.html", "r");
    if (!f) {
        httpd_resp_send_404(req);
        return ESP_FAIL;
    }

    char buf[256];
    size_t read_bytes;
    httpd_resp_set_type(req, "text/html");

    while ((read_bytes = fread(buf, 1, sizeof(buf), f)) > 0) {
        httpd_resp_send_chunk(req, buf, read_bytes);
    }
    fclose(f);
    httpd_resp_send_chunk(req, NULL, 0); // signal end
    return ESP_OK;
}

httpd_uri_t css = {
css.uri = "/style.css",
css.method = HTTP_GET,
css.handler = css_get_handler,
css.user_ctx = NULL
};
/* URI handler structure */
httpd_uri_t alternator = {
alternator.uri = "/alternator",
alternator.method = HTTP_GET,
alternator.handler = alternator_html_get_handler,
alternator.user_ctx = NULL
};
/* URI handler structure */
httpd_uri_t get_alternator = {
get_alternator.uri = "/getAlternator",
get_alternator.method = HTTP_GET,
get_alternator.handler = alternator_get_handler,
get_alternator.user_ctx = NULL
};
/* Start the HTTP server */
httpd_handle_t start_server(void)
{
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();
    httpd_handle_t server = NULL;

    if (httpd_start(&server, &config) == ESP_OK)
    {
        httpd_register_uri_handler(server, &alternator);
        httpd_register_uri_handler(server, &css);
        httpd_register_uri_handler(server, &get_alternator);
    }
    return server;
}

extern "C" void app_main(void)
{

    // Initialize NVS
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND)
    {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);

    // Initialize TCP/IP and Wi-Fi here
    ESP_ERROR_CHECK(esp_netif_init());
    init_spiffs();
    wifi_init();
    // Start HTTP server
    start_server();

    ESP_LOGI(TAG, "HTTP server started. Visit http://<ESP32_IP>/hello");
}
